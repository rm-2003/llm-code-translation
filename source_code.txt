import numpy as np
from scipy import integrate

# Create a NumPy array of 1000 points between 0 and 10
x = np.linspace(0, 10, 1000)

# Define a function to integrate: f(x) = sin(x) * exp(-x/3)
def f(x):
    return np.sin(x) * np.exp(-x/3)

# Compute the definite integral of f(x) from 0 to 10 using SciPy
result, error = integrate.quad(f, 0, 10)

print("Integral result:", result)
print("Estimated error:", error)
